cmake_minimum_required(VERSION 3.24)
project(kinoko_upstream_contracts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
include(../../cmake/Xiph.cmake)
add_executable(kinoko_vorbis_decoder_contract ../vorbis_decoder_contract.cpp)
target_link_libraries(kinoko_vorbis_decoder_contract PRIVATE kinoko_vorbis_decoder kinoko_vorbisenc)
add_test(NAME vorbis_decoder_contract COMMAND kinoko_vorbis_decoder_contract --snapshot "${CMAKE_BINARY_DIR}/vorbis-snapshot.bin")

set(KINOKO_SQUIRREL2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../third_party/squirrel-2.2.2")
include(../../cmake/Bindings.cmake)
file(GLOB KINOKO_CONTRACT_VM_SOURCES CONFIGURE_DEPENDS "${KINOKO_SQUIRREL2_ROOT}/squirrel/*.cpp")
add_library(kinoko_contract_vm STATIC ${KINOKO_CONTRACT_VM_SOURCES} direct_native_invoke.cpp)
target_include_directories(kinoko_contract_vm PUBLIC "${KINOKO_SQUIRREL2_ROOT}/include")
if(NOT MSVC)
    target_compile_options(kinoko_contract_vm PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/vm_allocator_declarations.hpp")
endif()
add_executable(kinoko_upstream_bindings_contract ../upstream_bindings_contract.cpp)
target_include_directories(kinoko_upstream_bindings_contract PRIVATE "${KINOKO_SQPLUS_DIR}")
target_link_libraries(kinoko_upstream_bindings_contract PRIVATE kinoko_upstream_bindings kinoko_contract_vm)
add_test(NAME upstream_bindings_contract COMMAND kinoko_upstream_bindings_contract)

include(../../cmake/BoostCount.cmake)
add_executable(kinoko_boost_control_contract ../boost_control_contract.cpp)
target_include_directories(kinoko_boost_control_contract PRIVATE "${KINOKO_BOOST_ROOT}")
target_link_libraries(kinoko_boost_control_contract PRIVATE kinoko_boost_count)
add_test(NAME boost_control_contract COMMAND kinoko_boost_control_contract)
set_tests_properties(boost_control_contract PROPERTIES TIMEOUT 120)
